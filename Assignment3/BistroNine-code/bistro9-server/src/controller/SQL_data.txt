-- 1. Create a new database for your project
CREATE DATABASE IF NOT EXISTS restaurant_db;

-- 2. Select this database to use it
USE restaurant_db;



-- 3. Now run your table creation

CREATE TABLE customer
(
	phoneNumber VARCHAR(100) NOT NULL,
    email VARCHAR(100),
    
    PRIMARY KEY (phoneNumber)
);

CREATE TABLE subscriber (
    id INT NOT NULL,          
    subscriberId INT  ,
    firstName VARCHAR(100) NOT NULL,
    lastName VARCHAR(100) NOT NULL,
    phoneNumber VARCHAR(100) NOT NULL UNIQUE,   
    type  ENUM('subscriber', 'restaurant representative', 'restaurant manager') NOT NULL,
    personalInfo VARCHAR(1000),
    username VARCHAR(100) UNIQUE,
    password VARCHAR(100), 
    
    PRIMARY KEY (id),
    FOREIGN KEY (phoneNumber) REFERENCES customer(phoneNumber) ON UPDATE CASCADE
);
CREATE TABLE restaurant_tables (
    tableId INT AUTO_INCREMENT ,
    seatsNumber INT NOT NULL, 
    location ENUM('inside', 'bar', 'outside') NOT NULL, 
    status ENUM('available', 'reserved', 'occupied') NOT NULL,
    
    PRIMARY KEY (tableId)
);
CREATE TABLE table_reservations (
    reservationID INT AUTO_INCREMENT,              
    reservationDate DATETIME NOT NULL,       
    numberOfDiners INT NOT NULL,             
    confirmationCode INT NOT NULL ,           
    phoneNumber VARCHAR(100) NOT NULL,              
    dateOfMakeReservation DATETIME DEFAULT CURRENT_TIMESTAMP, 
    arrivalTime DATETIME, 
	leavingTime DATETIME,
	tableId INT,
	status ENUM('active', 'cancelled', 'completed') NOT NULL ,
	
    PRIMARY KEY (reservationID),
    FOREIGN KEY (phoneNumber) REFERENCES customer(phoneNumber) ON UPDATE CASCADE,
    FOREIGN KEY (tableId) REFERENCES restaurant_tables(tableId)
);
